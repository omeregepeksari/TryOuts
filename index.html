<script>


const TILE = 46;

const canvas = document.getElementById('board');

const ctx = canvas.getContext('2d');


let grid = [];       // 'wall' | 'floor'

let targets = [];    // boolean grid

let crates = [];     // {x,y}

let player = { x:0, y:0 };

let rows = 0, cols = 0;

let history = [];    // stack of {crates, player, moves, pushes}

let moves = 0, pushes = 0;

let currentLevel = 0;

let bestByLevel = {};

let solvedByLevel = {};


function parseLevel(str) {

  const lines = str.split('\n');

  rows = lines.length;

  cols = Math.max(...lines.map(l => l.length));

  grid = []; targets = []; crates = [];

  for (let y = 0; y < rows; y++) {

    const line = lines[y];

    const gRow = [], tRow = [];

    for (let x = 0; x < cols; x++) {

      const ch = line[x] || ' ';

      let isWall = ch === '#';

      let isTarget = false;

      if (ch === '.' || ch === '+' || ch === '*') isTarget = true;

      if (ch === '@' || ch === '+') { player = { x, y }; }

      if (ch === '$' || ch === '*') { crates.push({ x, y }); }

      gRow.push(isWall ? 'wall' : (ch === ' ' ? 'void' : 'floor'));

      tRow.push(isTarget);

    }

    grid.push(gRow);

    targets.push(tRow);

  }

}


function loadLevel(idx) {

  currentLevel = idx;

  parseLevel(LEVELS[idx]);

  history = [];

  moves = 0; pushes = 0;

  canvas.width = cols * TILE;

  canvas.height = rows * TILE;

  updateHUD();

  document.getElementById('nextBtn').style.display = 'none';

  draw();

}


function isWall(x, y) {

  return !grid[y] || grid[y][x] === 'wall' || grid[y][x] === undefined;

}

function crateAt(x, y) {

  return crates.find(c => c.x === x && c.y === y);

}

function isTarget(x, y) {

  return targets[y] && targets[y][x];

}


function snapshot() {

  return {

    crates: crates.map(c => ({...c})),

    player: {...player},

    moves, pushes

  };

}


function tryMove(dx, dy) {

  const nx = player.x + dx, ny = player.y + dy;

  if (isWall(nx, ny)) return;

  const crate = crateAt(nx, ny);

  if (crate) {

    const bx = nx + dx, by = ny + dy;

    if (isWall(bx, by) || crateAt(bx, by)) return; // blocked push

    history.push(snapshot());

    crate.x = bx; crate.y = by;

    player.x = nx; player.y = ny;

    moves++; pushes++;

  } else {

    history.push(snapshot());

    player.x = nx; player.y = ny;

    moves++;

  }

  updateHUD();

  draw();

  checkWin();

}


function undo() {

  if (!history.length) return;

  const s = history.pop();

  crates = s.crates; player = s.player; moves = s.moves; pushes = s.pushes;

  updateHUD();

  draw();

}


function resetLevel() { loadLevel(currentLevel); }


function checkWin() {

  const solved = crates.every(c => isTarget(c.x, c.y));

  if (solved) {

    solvedByLevel[currentLevel] = true;

    if (!bestByLevel[currentLevel] || moves < bestByLevel[currentLevel]) {

      bestByLevel[currentLevel] = moves;

    }

    updateHUD();

    showToast(`Solved in ${moves} moves! 🎉`);

    document.getElementById('nextBtn').style.display = currentLevel < LEVELS.length - 1 ? 'inline-block' : 'none';


    const allSolved = LEVELS.every((_, i) => solvedByLevel[i]);

    if (allSolved) {

      const total = LEVELS.reduce((sum, _, i) => sum + bestByLevel[i], 0);

      setTimeout(() => showFinalScore(total), 1200);

    }

  }

}


// ---------------- Leaderboard flow ----------------

function showFinalScore(total) {

  document.getElementById('finalScoreLine').textContent =

    `All 12 levels cleared in ${total} total moves. Submit your score?`;

  document.getElementById('nameModal').classList.add('show');

}

document.getElementById('skipSubmit').onclick = () => {

  document.getElementById('nameModal').classList.remove('show');

};

document.getElementById('submitScore').onclick = async () => {

  const name = document.getElementById('nameInput').value.trim() || 'Anonymous';

  const total = LEVELS.reduce((sum, _, i) => sum + (bestByLevel[i] || 0), 0);

  document.getElementById('nameModal').classList.remove('show');

  if (typeof window.submitScore === 'function') {

    showToast('Submitting score…');

    try {

      await window.submitScore(name, total);

      showToast('Score submitted! 🎉');

    } catch (e) {

      showToast('Could not submit — check Firebase setup.');

    }

  }

  openLeaderboard();

};


function openLeaderboard() {

  document.getElementById('lbModal').classList.add('show');

  refreshLeaderboard();

}

async function refreshLeaderboard() {

  const statusEl = document.getElementById('lbStatus');

  const listEl = document.getElementById('lbList');

  listEl.innerHTML = '';

  if (typeof window.loadLeaderboard !== 'function') {

    statusEl.textContent = 'Leaderboard not configured yet.';

    return;

  }

  statusEl.textContent = 'Loading…';

  try {

    const rows = await window.loadLeaderboard();

    if (!rows.length) { statusEl.textContent = 'No scores yet — be the first!'; return; }

    statusEl.textContent = '';

    rows.forEach((r, i) => {

      const li = document.createElement('li');

      li.innerHTML = `<span>#${i+1} ${r.name}</span><span>${r.totalMoves} moves</span>`;

      listEl.appendChild(li);

    });

  } catch (e) {

    statusEl.textContent = 'Could not load leaderboard — check Firebase setup.';

  }

}

document.getElementById('lbBtn').onclick = openLeaderboard;

document.getElementById('closeLb').onclick = () => document.getElementById('lbModal').classList.remove('show');


function showToast(msg) {

  const t = document.getElementById('toast');

  t.textContent = msg;

  t.classList.add('show');

  setTimeout(() => t.classList.remove('show'), 2200);

}


function updateHUD() {

  document.getElementById('lvlNum').textContent = currentLevel + 1;

  document.getElementById('lvlTotal').textContent = LEVELS.length;

  document.getElementById('moveCount').textContent = moves;

  document.getElementById('pushCount').textContent = pushes;

  document.getElementById('bestMoves').textContent = bestByLevel[currentLevel] ?? '—';

}


// ---------------- Rendering ----------------

function draw() {

  ctx.clearRect(0, 0, canvas.width, canvas.height);

  for (let y = 0; y < rows; y++) {

    for (let x = 0; x < cols; x++) {

      const px = x*TILE, py = y*TILE;

      const cell = grid[y][x];

      if (cell === 'void') continue;

      if (cell === 'wall') {

        ctx.fillStyle = '#3a3f5c';

        ctx.fillRect(px, py, TILE, TILE);

        ctx.strokeStyle = 'rgba(0,0,0,0.25)';

        ctx.strokeRect(px+1, py+1, TILE-2, TILE-2);

        ctx.fillStyle = 'rgba(255,255,255,0.04)';

        ctx.fillRect(px+4, py+4, TILE-8, 6);

      } else {

        ctx.fillStyle = (x+y)%2===0 ? '#181b30' : '#1c1f38';

        ctx.fillRect(px, py, TILE, TILE);

        if (targets[y][x]) {

          ctx.strokeStyle = '#ffd166';

          ctx.lineWidth = 2;

          ctx.beginPath();

          ctx.arc(px+TILE/2, py+TILE/2, TILE*0.22, 0, Math.PI*2);

          ctx.stroke();

          ctx.fillStyle = 'rgba(255,209,102,0.25)';

          ctx.fill();

        }

      }

    }

  }


  // crates

  crates.forEach(c => {

    const px = c.x*TILE, py = c.y*TILE;

    const onTarget = isTarget(c.x, c.y);

    ctx.fillStyle = onTarget ? '#6fcf97' : '#d98a4f';

    ctx.strokeStyle = onTarget ? '#3fa768' : '#a5602c';

    ctx.lineWidth = 3;

    roundRect(px+5, py+5, TILE-10, TILE-10, 6);

    ctx.fill(); ctx.stroke();

    ctx.strokeStyle = 'rgba(0,0,0,0.2)';

    ctx.lineWidth = 1;

    ctx.strokeRect(px+11, py+11, TILE-22, TILE-22);

  });


  // player

  const pxp = player.x*TILE, pyp = player.y*TILE;

  ctx.fillStyle = '#7cc4ff';

  ctx.beginPath();

  ctx.arc(pxp+TILE/2, pyp+TILE/2, TILE*0.32, 0, Math.PI*2);

  ctx.fill();

  ctx.fillStyle = '#1a1d2e';

  ctx.beginPath();

  ctx.arc(pxp+TILE/2-6, pyp+TILE/2-4, 3.5, 0, Math.PI*2);

  ctx.arc(pxp+TILE/2+6, pyp+TILE/2-4, 3.5, 0, Math.PI*2);

  ctx.fill();

}


function roundRect(x, y, w, h, r) {

  ctx.beginPath();

  ctx.moveTo(x+r, y);

  ctx.arcTo(x+w, y, x+w, y+h, r);

  ctx.arcTo(x+w, y+h, x, y+h, r);

  ctx.arcTo(x, y+h, x, y, r);

  ctx.arcTo(x, y, x+w, y, r);

  ctx.closePath();

}


// ---------------- Hint (simple BFS solver for nudge, not full solve) ----------------

function hint() {

  // Find first crate not on a target, suggest the nearest target direction as a nudge.

  const stray = crates.find(c => !isTarget(c.x, c.y));

  if (!stray) { showToast('Everything is already placed!'); return; }

  let nearest = null, best = Infinity;

  for (let y = 0; y < rows; y++) for (let x = 0; x < cols; x++) {

    if (targets[y] && targets[y][x] && !crateAt(x,y)) {

      const d = Math.abs(x-stray.x)+Math.abs(y-stray.y);

      if (d < best) { best = d; nearest = {x,y}; }

    }

  }

  if (!nearest) { showToast('Think about which target needs this crate.'); return; }

  const dx = nearest.x - stray.x, dy = nearest.y - stray.y;

  const dir = Math.abs(dx) > Math.abs(dy) ? (dx>0?'right':'left') : (dy>0?'down':'up');

  showToast(`Try moving a crate ${dir} — nearest empty target is that way.`);

}


// ---------------- Level select ----------------

function buildLevelSelect() {

  const el = document.getElementById('levelSelect');

  el.innerHTML = '';

  LEVELS.forEach((_, i) => {

    const b = document.createElement('button');

    b.textContent = i+1;

    if (solvedByLevel[i]) b.classList.add('done');

    b.onclick = () => { loadLevel(i); toggleLevelSelect(false); };

    el.appendChild(b);

  });

}

function toggleLevelSelect(force) {

  const el = document.getElementById('levelSelect');

  const show = force !== undefined ? force : el.style.display !== 'grid';

  if (show) buildLevelSelect();

  el.style.display = show ? 'grid' : 'none';

}


// ---------------- Input ----------------

window.addEventListener('keydown', e => {

  const map = {

    ArrowUp:[0,-1], KeyW:[0,-1],

    ArrowDown:[0,1], KeyS:[0,1],

    ArrowLeft:[-1,0], KeyA:[-1,0],

    ArrowRight:[1,0], KeyD:[1,0]

  };

  if (map[e.code]) { e.preventDefault(); tryMove(...map[e.code]); }

  if (e.code === 'KeyZ') undo();

  if (e.code === 'KeyR') resetLevel();

});

document.getElementById('levelsBtn').onclick = () => toggleLevelSelect();

document.getElementById('nextBtn').onclick = () => {

  if (currentLevel < LEVELS.length - 1) loadLevel(currentLevel + 1);

};


// touch swipe support

let touchStart = null;

canvas.addEventListener('touchstart', e => {

  touchStart = { x: e.touches[0].clientX, y: e.touches[0].clientY };

}, {passive:true});

canvas.addEventListener('touchend', e => {

  if (!touchStart) return;

  const dx = e.changedTouches[0].clientX - touchStart.x;

  const dy = e.changedTouches[0].clientY - touchStart.y;

  if (Math.abs(dx) > Math.abs(dy)) {

    if (Math.abs(dx) > 20) tryMove(dx>0?1:-1, 0);

  } else {

    if (Math.abs(dy) > 20) tryMove(0, dy>0?1:-1);

  }

  touchStart = null;

});


loadLevel(0);

      db = getFirestore(app);

    }

  } catch (e) {

    console.warn('Firebase not configured yet:', e);

  }


  window.submitScore = async function(name, totalMoves) {

    if (!db) throw new Error('Firebase not configured');

    await addDoc(collection(db, 'scores'), {

      name: String(name).slice(0, 16),

      totalMoves: Math.round(totalMoves),

      createdAt: serverTimestamp()

    });

  };


  window.loadLeaderboard = async function() {

    if (!db) throw new Error('Firebase not configured');

    const q = query(collection(db, 'scores'), orderBy('totalMoves', 'asc'), limit(10));

    const snap = await getDocs(q);

    return snap.docs.map(d => d.data());

  };
